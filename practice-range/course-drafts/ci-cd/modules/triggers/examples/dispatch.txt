on:
  workflow_dispatch:
    inputs:
      environment:
        description: Where to deploy
        type: choice
        options: [staging, production]
        required: true
      version:
        description: Image tag to deploy, for example 1.5.0
        type: string
        required: true

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - run: |
          echo "Version:     ${{ inputs.version }}"
          echo "Environment: ${{ inputs.environment }}"
