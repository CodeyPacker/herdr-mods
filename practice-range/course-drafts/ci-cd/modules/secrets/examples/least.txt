name: Deploy

# The default for every job in this file: read the code, nothing else.
permissions:
  contents: read

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v5
      - run: npm ci && npm test

  deploy:
    needs: test
    runs-on: ubuntu-latest
    environment: production
    # Only this job may request an identity token.
    permissions:
      contents: read
      id-token: write
    steps:
      - run: echo "sign in and deploy here"
