jobs:
  build:
    runs-on: ubuntu-latest
    outputs:
      tag: ${{ steps.name.outputs.tag }}
    steps:
      - uses: actions/checkout@v5

      # Compute the name once. Seven characters of the commit identifier.
      - id: name
        run: echo "tag=git-${GITHUB_SHA::7}" >> "$GITHUB_OUTPUT"

  deploy-dev:
    needs: build
    runs-on: ubuntu-latest
    steps:
      - run: echo "Deploying tracker:${{ needs.build.outputs.tag }}"
