# Dockerfile
# The token is readable at /run/secrets/npm_token during this step only.
# It is not stored in the image.
RUN --mount=type=secret,id=npm_token \
    NPM_TOKEN="$(cat /run/secrets/npm_token)" npm ci --omit=dev

# Build. The token comes from a file outside the build context.
docker build --secret id=npm_token,src=$HOME/.npm_token -t tracker:1.4.2 .
