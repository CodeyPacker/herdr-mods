FROM node:latest
WORKDIR /app
COPY . .
ENV DATABASE_PASSWORD=example-only-not-a-real-password
RUN apt-get update
RUN apt-get install -y vim net-tools curl
RUN npm install
RUN npm run build
EXPOSE 3000
CMD npm start
