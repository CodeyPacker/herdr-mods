# See which platform an image was built for.
docker image inspect --format '{{.Os}}/{{.Architecture}}' tracker:1.5.0

# Build for one named platform.
docker build --platform linux/amd64 -t tracker:1.5.0 .

# Build for both and push them under one tag.
docker buildx build \
  --platform linux/amd64,linux/arm64 \
  -t registry.example.com/team/tracker:1.5.0 \
  --push .
